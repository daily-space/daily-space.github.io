rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isOwner() {
      return request.auth != null && request.auth.token.email == "lesya009ghh8@gmail.com";
    }
    match /{document=**} {
      allow read, write: if isOwner();
    }
    match /projects/{project}/access/{email} {
      allow read: if request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.email.lower() == email;
    }
    match /projects/{project}/users/{uid} {
      allow read: if request.auth != null
        && request.auth.uid == uid
        && request.auth.token.email_verified == true
        && exists(/databases/$(database)/documents/projects/$(project)/access/$(request.auth.token.email.lower()));
      allow write: if request.auth != null
        && request.auth.uid == uid
        && request.auth.token.email_verified == true
        && exists(/databases/$(database)/documents/projects/$(project)/access/$(request.auth.token.email.lower()))
        && request.resource.data.json.size() < 900000;
    }
  }
}
